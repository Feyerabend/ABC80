\T{v:ASABEG}\texttt{ASABEG} & \texttt{\$FDD4} & ASCII-aritmetik: a:s början\\
\T{v:ASAEND}\texttt{ASAEND} & \texttt{\$FDD6} & ASCII-aritmetik: a:s slut (\adrx{83})\\
\T{v:SINQ}\texttt{SINQ} & \texttt{\$FDE4} & \kw{SIN}/\kw{COS}: kvadranten\\
\T{v:ASARG1}\texttt{ASARG1} & \texttt{\$FDE7} & ASCII-aritmetik: a\$ i bufferten\\

\T{v:CLOCKH}\texttt{CLOCKH} & \texttt{\$FDF1} & \namnx{CLOCK}:s två höga byte (\adr{FDF1}–\adr{FDF2}), räknas ned var 256:e NMI\\
\T{v:CURROW}\texttt{CURROW} & \texttt{\$FDF3} & markörens rad (0–23)\\
\T{v:CURCOL}\texttt{CURCOL} & \texttt{\$FDF4} & markörens kolumn (0–39)\\
\T{v:KEYFLG}\texttt{KEYFLG} & \texttt{\$FDF5} & bit 7 = tangent väntar (sätts av \namn{KEYINT})\\
\T{v:RPTDLY}\texttt{RPTDLY} & \texttt{\$FDF7} & autorepetition: fördröjning\\

\T{v:EDPTR}\texttt{EDPTR} & \texttt{\$FDFB} & \namn{INLINE}: mallrad för TAB-tangenten\\
\T{v:CASBLK}\texttt{CASBLK} & \texttt{\$FDFD} & kassett: blocknummer\\
\T{v:RCVPTR}\texttt{RCVPTR} & \texttt{\$FDFF} & kassettläsning: sparat DE\\
\T{v:RCVBC}\texttt{RCVBC} & \texttt{\$FE01} & kassettläsning: C = skiftregister, B = bitar kvar\\
\T{v:RCVCONT}\texttt{RCVCONT} & \texttt{\$FE03} & kassettläsning: korutinens fortsättningsadress\\
\T{v:RCVSUM}\texttt{RCVSUM} & \texttt{\$FE05} & kassettläsning: checksumma\\
\T{v:BRKFLG}\texttt{BRKFLG} & \texttt{\$FE07} & skilt från 0 om BREAK (CTRL-C) har tryckts\\
\T{v:DEVHEAD}\texttt{DEVHEAD} & \texttt{\$FE0A} & första posten i enhetslistan\\

\T{v:BOFA}\texttt{BOFA} & \texttt{\$FE1C} & programmets början\\

\T{v:STKTOP}\texttt{STKTOP} & \texttt{\$FE27} & stackens startvärde\\
\T{v:VARLST}\texttt{VARLST} & \texttt{\$FE29} & variabellistan\\
\texttt{VFE2B} & \texttt{\$FE2B} & kortvalet: senaste värdet till port 1, skrivs till port 1 igen före varje \kw{OUT} och \kw{INP} (IY+\adrx{15})\\
\T{v:CURLIN}\texttt{CURLIN} & \texttt{\$FE2C} & adress till raden som körs (eller hade fel)\\
\T{v:DATPTR}\texttt{DATPTR} & \texttt{\$FE2E} & \namn{DATA}: pekare till nästa värde\\
\T{v:DATLIN}\texttt{DATLIN} & \texttt{\$FE30} & \namn{DATA}: aktuell rad\\
\T{v:FILLST}\texttt{FILLST} & \texttt{\$FE32} & fillistan: första FCB i kedjan (IY+\adrx{1C})\\
\T{v:VFE33}\texttt{VFE33} & \texttt{\$FE33} & \namn{FILLST}:s höga byte, 0 = inga filer (IY+\adrx{1D})\\
\T{v:ONERR}\texttt{ONERR} & \texttt{\$FE34} & \kw{ON ERROR GOTO}: radadress, 0 = inget (IY+\adrx{1E}/\adrx{1F})\\
\T{v:STMSP}\texttt{STMSP} & \texttt{\$FE36} & SP när satsen startade, för \kw{ON ERROR}\\
\T{v:FNFRAM}\texttt{FNFRAM} & \texttt{\$FE3E} & ramen för aktuellt \kw{FN}-anrop (argumenten)\\
